// AM_INIT
//----------------------------------------------------------------------------
macro AM_INIT
  Routine QRY_<queryNo>
  // initialization   
  EO_INIT             
  EO_ENTER            
  EO_SQL <queryNo>

  // check for ZK_STARTMARKER
  EO_DUP                          
  EO_INTEGER ZK_STARTMARKER       
  EO_EQUAL                        
  EO_JMP_TRUE STARTMARKER_HANDLER 

  // check for ZK_DATAPULSE
  EO_DUP                        
  EO_INTEGER ZK_DATAPULSE       
  EO_EQUAL                      
  EO_JMP_TRUE DATAPULSE_HANDLER 

  // check for ZK_STREAMDEF
  EO_DUP                        
  EO_INTEGER  ZK_STREAMDEF      
  EO_EQUAL                      
  EO_JMP_TRUE STREAMDEF_HANDLER 

  // check for ZK_CRWBUFFER
  EO_DUP                        
  EO_INTEGER ZK_CRWBUFFER       
  EO_EQUAL                      
  EO_JMP_TRUE CRWBUFFER_HANDLER 

  // check for ZK_EOSMARKER
  EO_DUP                        
  EO_INTEGER  ZK_EOSMARKER      
  EO_EQUAL                      
  EO_JMP_TRUE EOSMARKER_HANDLER 

  // Never reach
  EO_PRINTF "**NO_HANDLER**\n"
  EO_NEVERREACH                       

endMacro

//---------------------------------------------------------------------------
// BEGIN STREAMDEF_HANDLER

macro AM_STREAMDEF_HANDLER_BEGIN
  // STREAMDEF_HANDLER
  Label STREAMDEF_HANDLER
  EO_PRINTF "**STREAMDEF_HANDLER**\n"
  EO_POP
  EO_REG_STREAMDEFOUT_NEW
endMacro

//----------------------------------------------------------------------------
// ADD COLUMN TO STREAMDEF
//----------------------------------------------------------------------------
macro AM_STREAMDEF_ADD_COLUMN
  EO_PUSH_COLUMN_SCALE <C_TAG_COLUMN_SCALE>
  EO_PUSH_COLUMN_LENGTH <C_TAG_COLUMN_LENGTH>
  EO_PUSH_COLUMN_TYPE <C_TAG_COLUMN_TYPE>
  EO_REG_STREAMDEFOUT_ADDCOLUMN <C_TAG_COLUMN_NAME>
 endMacro

//----------------------------------------------------------------------------
// END STREAMDEF_HANDLER
//----------------------------------------------------------------------------
macro AM_STREAMDEF_HANDLER_END
  EO_POT_USE_STREAMDEFOUT 0 
  EO_JMP QRY_DONE
endMacro

//----------------------------------------------------------------------------
// CRWBUFFER_HANDLER
//----------------------------------------------------------------------------
macro AM_CRWBUFFER_HANDLER
  // CRWBUFFER_HANDLER
  Label CRWBUFFER_HANDLER
  EO_PRINTF "**CRWBUFFER_HANDLER**\n"
  EO_POP                                     
  EO_CRW_SRC_BEGIN                           

  // NEXT_CRW_SRC
  Label NEXT_CRW_SRC
  EO_CRW_SRC_NEXT             
  EO_JMP_FALSE QRY_DONE       

  EO_CALL <C_TAG_CRW_PROCESS_ROUTINE>
  EO_JMP NEXT_CRW_SRC       
  EO_JMP QRY_DONE
endMacro

 PROJECT_COLUMNS
//----------------------------------------------------------------------------
// AM_PROJECT_COLUMNS(iSrcColumn, iDstColumn)
macro AM_PROJECT_COLUMNS
  EO_SET_COLUMN <C_TAG_PROJECT_SRC_COLUMN >
  EO_COL_SRC_PUSHSET      
  EO_RSB_DST_POPCOLI <C_TAG_PROJECT_DST_COLUMN>
endMacro

//----------------------------------------------------------------------------
 END PROJECT_COLUMNS
//----------------------------------------------------------------------------
// AM_PROJECT_COLUMNS_END
macro AM_PROJECT_COLUMNS_END

endMacro
