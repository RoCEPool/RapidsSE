macro AM_INIT
//----------------------------------------------------------------------------//
// macro AM_INIT                                                              //
//----------------------------------------------------------------------------//
  Routine QRY_<queryNo>
  // initialization   
  EO_INIT             
  EO_ENTER            
  EO_SQL <subQueryNo>

  // check for ZK_STARTMARKER
  EO_DUP                          
  EO_INTEGER ZK_STARTMARKER       
  EO_EQUAL                        
  EO_JMP_TRUE STARTMARKER_HANDLER 

  // check for ZK_DATAPULSE
  EO_DUP                        
  EO_INTEGER ZK_DATAPULSE       
  EO_EQUAL                      
  EO_JMP_TRUE DATAPULSE_HANDLER 

  // check for ZK_STREAMDEF
  EO_DUP                        
  EO_INTEGER  ZK_STREAMDEF      
  EO_EQUAL                      
  EO_JMP_TRUE STREAMDEF_HANDLER 

  // check for ZK_CRWBUFFER
  EO_DUP                        
  EO_INTEGER ZK_CRWBUFFER       
  EO_EQUAL                      
  EO_JMP_TRUE CRWBUFFER_HANDLER 

  // check for ZK_EOSMARKER
  EO_DUP                        
  EO_INTEGER  ZK_EOSMARKER      
  EO_EQUAL                      
  EO_JMP_TRUE EOSMARKER_HANDLER 

  // Never reach
  EO_PRINTF "**NO_HANDLER**\n"
  EO_NEVERREACH                       

endMacro

macro AM_CRWBUFFER_PROJECT_HANDLER
//----------------------------------------------------------------------------//
// macro AM_CRWBUFFER_PROJECT_HANDLER                                         //
//----------------------------------------------------------------------------//
  // CRWBUFFER_PROJECT_HANDLER
  Label CRWBUFFER_HANDLER
  EO_PRINTF "**CRWBUFFER_HANDLER**\n"
  EO_POP
  EO_CRW_SRC_BEGIN

  // NEXT_CRW_SRC
  Label NEXT_CRW_SRC
  EO_CRW_SRC_NEXT
  EO_JMP_FALSE QRY_DONE
  EO_CALL <crwProcessRoutine>
  EO_JMP NEXT_CRW_SRC
  EO_JMP QRY_DONE
 endMacro

macro AM_CRWBUFFER_FILTER_HANDLER
//----------------------------------------------------------------------------//
// macro AM_CRWBUFFER_FILTER_HANDLER                                          //
//----------------------------------------------------------------------------//
  // AM_CRWBUFFER_FILTER_HANDLER
  Label CRWBUFFER_HANDLER
  EO_PRINTF "**CRWBUFFER_HANDLER**\n"
  EO_POP
  EO_CRW_SRC_BEGIN

  // NEXT_CRW_SRC
  Label NEXT_CRW_SRC
  EO_CRW_SRC_NEXT
  EO_JMP_FALSE QRY_DONE

  EO_CALL <crwFilterRoutine>
  EO_JMP_FALSE NEXT_CRW_SRC

  EO_CALL <crwProcessRoutine>
  EO_JMP NEXT_CRW_SRC

  EO_JMP QRY_DONE
 endMacro

macro AM_FILTER_VARCHAR_COLUMN
  // FILTER_COLUMN
  //EO_PUSH_LOCALI <predicateSrcColumn>
  EO_STR_LITERAL <literalValue>
  <relationalOp>

  EO_JMP FILTER_VARCHAR_DONE

  Label FILTER_VARCHAR_TRUE
  EO_PUSH_1

  Label FILTER_VARCHAR_DONE
  EO_RETURN
endMacro
