// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN : QRY_300
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Routine QRY_300

  // initialization
  EO_INIT
  EO_ENTER

  // set sql query no (debug)
  EO_SQL 300

  // check for ZK_STARTMARKER
  EO_DUP
  EO_INTEGER ZK_STARTMARKER
  EO_EQUAL
  EO_JMP_TRUE STARTMARKER_HANDLER

  // check for ZK_PULSEMARKER
  EO_DUP
  EO_INTEGER ZK_PULSEMARKER
  EO_EQUAL
  EO_JMP_TRUE PULSEMARKER_HANDLER

  // check for ZK_STREAMDEF
  EO_DUP
  EO_INTEGER  ZK_STREAMDEF
  EO_EQUAL
  EO_JMP_TRUE STREAMDEF_HANDLER

  // check for ZK_CRWBUFFER
  EO_DUP
  EO_INTEGER ZK_CRWBUFFER
  EO_EQUAL
  EO_JMP_TRUE CRWBUFFER_HANDLER

  // check for ZK_EOSMARKER
  EO_DUP
  EO_INTEGER  ZK_EOSMARKER
  EO_EQUAL
  EO_JMP_TRUE EOSMARKER_HANDLER

  // Never reach
  EO_ASSERT

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_STARTMARKER handler
Label STARTMARKER_HANDLER
  EO_POP
  EO_TABLENM_SRC_SET "customer"
  EO_INDEXNM_SRC_SET "customer_Idx1"
  EO_TABLENM_ALT_SET "orders"
  EO_INDEXNM_ALT_SET "orders_Idx2"
  EO_JMP QRY_300_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_PULSEMARKER handler
Label PULSEMARKER_HANDLER
EO_POP
EO_JMP QRY_100_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_STREAMDEF handler
Label STREAMDEF_HANDLER
EO_POP

EO_REG_STREAMDEFOUT_NEW

// add column L_RETURNFLAG
EO_PUSH_COLUMN_SCALE -1
EO_PUSH_COLUMN_LENGTH 1
EO_PUSH_COLUMN_TYPE CT_VARCHAR
EO_REG_STREAMDEFOUT_ADDCOLUMN "l_returnflag"

// add column L_LINESTATUS
EO_PUSH_COLUMN_SCALE -1
EO_PUSH_COLUMN_LENGTH 1
EO_PUSH_COLUMN_TYPE CT_VARCHAR
EO_REG_STREAMDEFOUT_ADDCOLUMN "l_linestatus"

// add column SUM(L_QUANTITY) as SUM_QTY
EO_PUSH_COLUMN_SCALE 2
EO_PUSH_COLUMN_LENGTH 15
EO_PUSH_COLUMN_TYPE CT_DECIMAL
EO_REG_STREAMDEFOUT_ADDCOLUMN "sum_qty"

// add column SUM(L_EXTENDEDPRICE) as SUM_BASE_PRICESUM_QTY
EO_PUSH_COLUMN_SCALE 2
EO_PUSH_COLUMN_LENGTH 15
EO_PUSH_COLUMN_TYPE CT_DECIMAL
EO_REG_STREAMDEFOUT_ADDCOLUMN "sum_base_price"

// add column SUM(L_EXTENDEDPRICE*(1-L_DISCOUNT)) as SUM_DISC_PRICE
EO_PUSH_COLUMN_SCALE 2
EO_PUSH_COLUMN_LENGTH 15
EO_PUSH_COLUMN_TYPE CT_DECIMAL
EO_REG_STREAMDEFOUT_ADDCOLUMN "sum_base_price"

// add column SUM(L_EXTENDEDPRICE*(1-L_DISCOUNT)*(1+L_TAX)) as SUM_CHARGE
EO_PUSH_COLUMN_SCALE 2
EO_PUSH_COLUMN_LENGTH 15
EO_PUSH_COLUMN_TYPE CT_DECIMAL
EO_REG_STREAMDEFOUT_ADDCOLUMN "sum_charge"

// add column COUNT(L_QUANTITY) as COUNT_QTY
EO_PUSH_COLUMN_SCALE 0
EO_PUSH_COLUMN_LENGTH 19
EO_PUSH_COLUMN_TYPE CT_INTEGER
EO_REG_STREAMDEFOUT_ADDCOLUMN "count_qty"

// add column COUNT(L_EXTENDEDPRICE)
EO_PUSH_COLUMN_SCALE 0
EO_PUSH_COLUMN_LENGTH 19
EO_PUSH_COLUMN_TYPE CT_INTEGER
EO_REG_STREAMDEFOUT_ADDCOLUMN "count_price"

// add column SUM(L_DISCOUNT)) as SUM_DISCOUNT
EO_PUSH_COLUMN_SCALE 2
EO_PUSH_COLUMN_LENGTH 15
EO_PUSH_COLUMN_TYPE CT_DECIMAL
EO_REG_STREAMDEFOUT_ADDCOLUMN "sum_discount"

// add column COUNT(L_DISCOUNT)
EO_PUSH_COLUMN_SCALE 0
EO_PUSH_COLUMN_LENGTH 19
EO_PUSH_COLUMN_TYPE CT_INTEGER
EO_REG_STREAMDEFOUT_ADDCOLUMN "count_discount"

// add column COUNT(*) as COUNT_ORDER
EO_PUSH_COLUMN_SCALE 0
EO_PUSH_COLUMN_LENGTH 19
EO_PUSH_COLUMN_TYPE CT_INTEGER
EO_REG_STREAMDEFOUT_ADDCOLUMN "count_order"

EO_POT_USE_STREAMDEFOUT 0

EO_JMP QRY_300_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_EOSMARKER handler
Label EOSMARKER_HANDLER
EO_POP
EO_CALL SUBROUTINE_IXT_TO_RSB
EO_REG_INDEXTREE_DELETE
EO_JMP QRY_300_DONE





  EO_PUSH_0
  EO_CTR_0_POP


  // test if rtn has been called again
  EO_CALLEDAGAIN
  EO_JMP_TRUE,"PRC_CUS

  EO_INTEGRITY;


  // iterate all CUSTOMER
  EO_IXT_SRC_BEGIN
  EO_JMP_FALSE QRY_300_DONE

Label GOT_CUS

  EO_CTR_0_INC
  EO_CTR_0_PUSH
  EO_INTEGER,10000
  EO_LESSTHAN
  EO_JMP_TRUE PRC_CUS

  EO_PRINTF "**QRY_300 TO BE CALLED AGAIN**\n"
  EO_CALLAGAIN
  EO_EXIT
  EO_RETURN

Label PRC_CUS
  EO_CRW_SRC_DEREF

  EO_COL_SRC_PUSHONE_a,C_MKTSEGMENT
  EO_STR_LITERAL "BUILDING"
  EO_STR_COMPARE_a
  EO_JMP_NOTEQUAL NXT_CUSTOMER

  EO_CALL JOI_ORD


// next CUSTOMER
Label NXT_CUSTOMER
  EO_IXT_SRC_NEXT
  EO_JMP_TRUE GOT_CUS

// end of CUSTOMER
Label QRY_300_DONE
  EO_EXIT
  EO_RETURN

// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END ROUTINE: QRY_300                                           Q300
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: JOI_ORD  -- join with orders                       Q300 JOI_ORD
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label JOI_ORD
  EO_IXT_POS_ALTBYSRC
  EO_JMP_FALSE RET_ORD

Label NXT_ORD
  EO_IXT_ALT_NEXT

  EO_JMP_FALSE RET_ORD
  EO_IXT_CMP_SRCALT
  EO_JMP_FALSE RET_ORD

  EO_CRW_ALT_DEREF

  EO_COL_ALT_PUSHONE,O_ORDERDATE
  EO_DATE, "1995-03-15"
  EO_LESSTHAN
  EO_JMP_FALSE NXT_ORD

  EO_CALL,"DST_PUT

  EO_JMP NXT_ORD

Label RET_ORD
  EO_RETURN
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: JOI_ORD                                        Q300 JOI_ORD
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: DST_PUT                                            Q300 DST_PUT
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label DST_PUT

  // calc i_Ptn
  EO_PTN_RESET
  EO_COL_ALT_PUSHONE,O_ORDERKEY
  EO_PTN_RUNNING
  EO_PTN_CALC

  EO_SET_ITEMS,3
  EO_SET_COLUMN,O_SHIPPRIORITY  // 7
  EO_SET_COLUMN,O_ORDERDATE     // 4
  EO_SET_COLUMN,O_ORDERKEY      // 0
  EO_COL_ALT_PUSHSET

  EO_RSB_DST_NEW

  EO_RSB_DST_POPCOLI,Q300_SHIPPRIORITY
  EO_RSB_DST_POPCOLI,Q300_ORDERDATE
  EO_RSB_DST_POPCOLI,Q300_ORDERKEY


  EO_RETURN

// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: DST_PUT                                        Q300 DST_PUT
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
