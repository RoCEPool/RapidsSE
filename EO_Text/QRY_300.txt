// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN : QRY_300
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Routine QRY_300

  // initialization
  EO_INIT
  EO_ENTER

  // set sql query no (debug)
  EO_SQL 300

  // check for ZK_STARTMARKER
  EO_DUP
  EO_INTEGER ZK_STARTMARKER
  EO_EQUAL
  EO_JMP_TRUE STARTMARKER_HANDLER

  // check for ZK_PULSEMARKER
  EO_DUP
  EO_INTEGER ZK_PULSEMARKER
  EO_EQUAL
  EO_JMP_TRUE PULSEMARKER_HANDLER

  // check for ZK_STREAMDEF
  EO_DUP
  EO_INTEGER  ZK_STREAMDEF
  EO_EQUAL
  EO_JMP_TRUE STREAMDEF_HANDLER

  // check for ZK_CRWBUFFER
  EO_DUP
  EO_INTEGER ZK_CRWBUFFER
  EO_EQUAL
  EO_JMP_TRUE CRWBUFFER_HANDLER

  // check for ZK_EOSMARKER
  EO_DUP
  EO_INTEGER  ZK_EOSMARKER
  EO_EQUAL
  EO_JMP_TRUE EOSMARKER_HANDLER

  // Never reach
  EO_ASSERT

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_STARTMARKER handler
Label STARTMARKER_HANDLER
  EO_POP
  EO_TABLENM_SRC_SET "customer"
  EO_INDEXNM_SRC_SET "customer_Idx1"
  EO_TABLENM_ALT_SET "orders"
  EO_INDEXNM_ALT_SET "orders_Idx2"

  EO_REG_STREAMDEFOUT_NEW

  // add column L_RETURNFLAG
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderkey"

  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 0
  EO_PUSH_COLUMN_TYPE CT_DATE
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderdate"

  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_shippriority"

  EO_POT_USE_STREAMDEFOUT 0

  EO_JMP QRY_300_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_STREAMDEF handler
Label STREAMDEF_HANDLER
  EO_POP
  EO_JMP QRY_300_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_EOSMARKER handler
Label EOSMARKER_HANDLER
  EO_POP
  EO_CALL SUBROUTINE_IXT_TO_RSB
  EO_REG_INDEXTREE_DELETE
  EO_JMP QRY_300_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_PULSEMARKER handler
Label PULSEMARKER_HANDLER
  EO_POP

  EO_PUSH_0
  EO_CTR_0_POP

  // test if rtn has been called again
  EO_CALLEDAGAIN
  EO_JMP_TRUE PRC_CUS

  //EO_INTEGRITY;


  // iterate all CUSTOMER
  EO_IXT_SRC_BEGIN
  EO_JMP_FALSE QRY_300_DONE

Label GOT_CUS

  EO_CTR_0_INC
  EO_CTR_0_PUSH
  EO_INTEGER 10000
  EO_LESSTHAN
  EO_JMP_TRUE PRC_CUS

  EO_PRINTF "**QRY_300 TO BE CALLED AGAIN**\n"
  EO_CALLAGAIN
  EO_EXIT
  EO_RETURN

Label PRC_CUS
  EO_CRW_SRC_DEREF

  EO_COL_SRC_PUSHONE_a 6 //C_MKTSEGMENT
  EO_STR_LITERAL "BUILDING"
  EO_STR_COMPARE_a
  EO_JMP_NOTEQUAL NXT_CUSTOMER

  EO_CALL JOI_ORD


// next CUSTOMER
Label NXT_CUSTOMER
  EO_IXT_SRC_NEXT
  EO_JMP_TRUE GOT_CUS

// end of CUSTOMER
Label QRY_300_DONE
  EO_EXIT
  EO_RETURN

// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END ROUTINE: QRY_300                                           Q300
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: JOI_ORD  -- join with orders                       Q300 JOI_ORD
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label JOI_ORD
  EO_IXT_POS_ALTBYSRC
  EO_JMP_FALSE RET_ORD

Label NXT_ORD
  EO_IXT_ALT_NEXT

  EO_JMP_FALSE RET_ORD
  EO_IXT_CMP_SRCALT
  EO_JMP_FALSE RET_ORD

  EO_CRW_ALT_DEREF

  EO_COL_ALT_PUSHONE 4 //O_ORDERDATE
  EO_DATE  "1995-03-15"
  EO_LESSTHAN
  EO_JMP_FALSE NXT_ORD

  EO_CALL DST_PUT

  EO_JMP NXT_ORD

Label RET_ORD
  EO_RETURN
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: JOI_ORD                                        Q300 JOI_ORD
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: DST_PUT                                            Q300 DST_PUT
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label DST_PUT

  // calc i_Ptn
  EO_PTN_RESET
  EO_COL_ALT_PUSHONE 0 //O_ORDERKEY
  EO_PTN_RUNNING
  EO_PTN_CALC

  EO_SET_ITEMS 3
  EO_SET_COLUMN 7 //O_SHIPPRIORITY  // 7
  EO_SET_COLUMN 4 //O_ORDERDATE     // 4
  EO_SET_COLUMN 0 //O_ORDERKEY      // 0
  EO_COL_ALT_PUSHSET

  EO_RSB_DST_NEW

  EO_RSB_DST_POPCOLI 2 //Q300_SHIPPRIORITY
  EO_RSB_DST_POPCOLI 1 //Q300_ORDERDATE
  EO_RSB_DST_POPCOLI 0 //Q300_ORDERKEY

  //EO_RSB_DST_PRINT
  EO_RETURN

// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: DST_PUT                                        Q300 DST_PUT
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
