// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN: QRY_601
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Routine QRY_601

  EO_INIT
  EO_ENTER

  // check for PULSEMARKER
  EO_DUP
  EO_INTEGER  ZK_PULSEMARKER
  EO_EQUAL
  EO_JMP_TRUE PULSEMARKER_HANDLER

  // check for ZK_STARTMARKER
  EO_DUP
  EO_INTEGER  ZK_STARTMARKER
  EO_EQUAL
  EO_JMP_TRUE STARTMARKER_HANDLER

  // check for ZK_EOSMARKER
  EO_DUP
  EO_INTEGER ZK_EOSMARKER
  EO_EQUAL
  EO_JMP_TRUE EOSMARKER_HANDLER

  // ASSERT that its an RSBBUFFER
  EO_INTEGER ZK_RSBBUFFER
  EO_EQUAL
  EO_ASSERT

  EO_RSB_DST_TESTROW
  EO_JMP_TRUE RSB_SRC_BEGIN
  EO_RSB_DST_NEW
  EO_PUSH_0
  EO_RSB_DST_POPCOLI 0

Label RSB_SRC_BEGIN
  EO_RSB_SRC_BEGIN

Label NEXT_RSB_SRC
  EO_RSB_SRC_NEXT
  EO_JMP_FALSE QRY_601_DONE
  EO_CALL SUBROUTINE_AGGREGATE
  EO_JMP NEXT_RSB_SRC

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_STARTMARKER handler
Label STARTMARKER_HANDLER
  EO_POP
  EO_JMP QRY_601_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_EOSMARKER handler
Label EOSMARKER_HANDLER
  EO_POP
  EO_JMP QRY_601_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_PULSEMARKER handler
Label PULSEMARKER_HANDLER
  EO_POP

Label QRY_601_DONE
  EO_RETURN

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// END: QRY_601
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN: SUBROUTINE_AGGREGATE
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label SUBROUTINE_AGGREGATE
  EO_RSB_SRC_PUSHCOLI 0
  EO_RSB_DST_PUSHCOLI 0
  EO_ADD
  EO_RSB_DST_POPCOLI 0
  EO_RETURN
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// END: SUBROUTINE_AGGREGATE
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
