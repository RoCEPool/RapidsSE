// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN : QRY_302
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Routine QRY_302

  // initialization
  EO_INIT
  EO_ENTER

  // set sql query no (debug)
  EO_SQL 302

  // check for ZK_STARTMARKER
  EO_DUP
  EO_INTEGER ZK_STARTMARKER
  EO_EQUAL
  EO_JMP_TRUE STARTMARKER_HANDLER

  // check for ZK_PULSEMARKER
  EO_DUP
  EO_INTEGER ZK_PULSEMARKER
  EO_EQUAL
  EO_JMP_TRUE PULSEMARKER_HANDLER

  // check for ZK_STREAMDEF
  EO_DUP
  EO_INTEGER  ZK_STREAMDEF
  EO_EQUAL
  EO_JMP_TRUE STREAMDEF_HANDLER

  // check for ZK_RSBBUFFER
  EO_DUP
  EO_INTEGER ZK_RSBBUFFER
  EO_EQUAL
  EO_JMP_TRUE RSBBUFFER_HANDLER

  // check for ZK_EOSMARKER
  EO_DUP
  EO_INTEGER  ZK_EOSMARKER
  EO_EQUAL
  EO_JMP_TRUE EOSMARKER_HANDLER

  // Never reach
  EO_ASSERT

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_STARTMARKER handler
Label STARTMARKER_HANDLER
  EO_PRINTF "**STARTMARKER_HANDLER**\n"
  EO_POP
  EO_REG_INDEXDEF_NEW "idx_int"
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_INDEXDEF_ADDCOLUMN "intA"
  EO_REG_INDEXTREE_NEW
  EO_REG_INDEXDEF_DELETE
  EO_INDEXTREE_TMP_SET
  EO_INDEXTREE_DEV_SET 0
  EO_JMP QRY_302_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_PULSEMARKER handler
Label PULSEMARKER_HANDLER
  EO_PRINTF "**PULSEMARKER_HANDLER**\n"
  EO_POP
  EO_JMP QRY_302_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_STREAMDEF handler
Label STREAMDEF_HANDLER
  EO_PRINTF "**STREAMDEF_HANDLER**\n"
  EO_POP

  EO_REG_STREAMDEFOUT_NEW

  // add column l_orderkey
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "l_orderkey"

  // add column revenue
  EO_PUSH_COLUMN_SCALE 2
  EO_PUSH_COLUMN_LENGTH 17
  EO_PUSH_COLUMN_TYPE CT_DECIMAL
  EO_REG_STREAMDEFOUT_ADDCOLUMN "revenue"

  // add column o_orderdate
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 0
  EO_PUSH_COLUMN_TYPE CT_DATE
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderdate"

  // add column o_shippriority
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_shippriority"


  EO_POT_USE_STREAMDEFOUT 0

  EO_JMP QRY_302_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_EOSMARKER handler
Label EOSMARKER_HANDLER
  EO_PRINTF "**EOSMARKER_HANDLER**\n"
  EO_POP
  EO_CALL OUT_AGG
  EO_JMP QRY_END
  EO_JMP QRY_302_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_RSBBUFFER handler
Label RSBBUFFER_HANDLER
  EO_PRINTF "**RSBBUFFER_HANDLER**\n"

  // --
  EO_RSB_SRC_BEGIN

Label NXT_SRC
  EO_RSB_SRC_NEXT
  EO_JMP_FALSE QRY_302_DONE

  //EO_CALL,"CPY_DST
  EO_CALL LIM_SBR

  EO_JMP NXT_SRC


// --
Label QRY_302_DONE
EO_EXIT
EO_RETURN

// --
Label QRY_END

EO_EXIT
EO_RETURN
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END ROUTINE: QRY_302                                           Q302
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: LIM_SBR  --                                        Q302 LIM_SBR
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label LIM_SBR
  EO_PRINTF "**LIM_SBR**\n"
  //EO_IXT_TMP_ENTRIES
  //EO_PRINT

  EO_IXT_TMP_ENTRIES
  EO_INTEGER 10
  EO_LESSTHAN
  EO_JMP_TRUE LIM_INS

  // -- -- -- -- -- -- -- -- --
Label LIM_TST

  EO_IXT_TRV_BEGIN
  EO_JMP_FALSE LIM_RET // need an assert

  EO_RSB_SRC_PUSHCOLI 1 //Q301_REVENUE
  EO_ENT_PUSHELMI0
  EO_LESSEQUAL
  EO_JMP_TRUE LIM_RET

  EO_IXT_DEL_ENT


  // -- -- -- -- -- -- -- -- --
Label LIM_INS
  EO_PRINTF "LIM_INS\n"

  EO_ENTWRK_RESET

  EO_RSB_SRC_PUSHCOLI 1 // Q301_REVENUE
  //EO_DUP
  //EO_PRINT
  EO_ENTWRK_APPENDELMI

  EO_ENTWRK_CREATEPAY 24 //3 * C_BYTES_PER_WORD

  EO_IXT_INS_ENT

  EO_PAY_ADDRESS

  EO_RSB_SRC_PUSHCOLI 0 //Q301_ORDERKEY
  EO_PAY_STORE_COLI 0
  EO_RSB_SRC_PUSHCOLI 2 //Q301_ORDERDATE
  EO_PAY_STORE_COLI 1
  EO_RSB_SRC_PUSHCOLI 3 //Q301_SHIPPRIORITY
  EO_PAY_STORE_COLI 2


  // -- -- -- -- -- -- -- -- --
Label LIM_RET
  //EO_IXT_DUMP,(i8)0
  EO_RETURN
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: QRY_302                                        Q302 LIM_SBR
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: OUT_AGG  --                                        Q302 OUT_AGG
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label OUT_AGG
  EO_PRINTF "OUT_AGG\n"

  EO_IXT_TRV_BEGIN
  EO_JMP_FALSE TRV_RET

  Label TRV_ECH
  EO_PRINTF "TRV_ECH\n"

  EO_RSB_DST_NEW

  EO_PAY_ADDRESS

  EO_ENT_PUSHELMI0
  EO_RSB_DST_POPCOLI 1 // Q302_REVENUE

  EO_PAY_FETCH_COLI 0
  EO_RSB_DST_POPCOLI 0 // Q302_ORDERKEY
  EO_PAY_FETCH_COLI 1
  EO_RSB_DST_POPCOLI 2 // Q302_ORDERDATE
  EO_PAY_FETCH_COLI 2
  EO_RSB_DST_POPCOLI 3 // Q302_SHIPPRIORITY

  EO_RSB_DST_PRINT

Label TRV_NXT
  EO_IXT_TRV_NEXT
  EO_JMP_TRUE TRV_ECH

Label TRV_RET
  EO_PRINTF "TRV_RET\n"
  //EO_BREAKPOINT
  EO_RETURN
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: QRY_302                                        Q302 OUT_AGG
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
