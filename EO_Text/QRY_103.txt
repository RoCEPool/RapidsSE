Routine QRY_103

EO_INIT
EO_ENTER

EO_JMP_TRUE BEGIN_RSB_SRC

// handle EOSMarker
EO_CALL SUBROUTINE_EOS

EO_JMP QRY_103_DONE

Label BEGIN_RSB_SRC
//vAddCodeString(EO_PRINTF **103** Got RSB BUFFER\n
// -- -- -- -- -- -- -- -- -- -- --
// insert each rsb

// forearch src rsb
EO_RSB_SRC_BEGIN

Label NEXT_RSB_SRC

EO_RSB_SRC_NEXT
EO_JMP_FALSE QRY_103_DONE

EO_RSB_SRC_PRINT

// reset entwrk
EO_ENTWRK_RESET

//ray100
// add key L_RETURNFLAG, L_LINESTATUS
//  EO_ENTWRK_APPENDELM, (i8)Q101_RETURNFLAG
//  EO_ENTWRK_APPENDELM, (i8)Q101_LINESTATUS
EO_KEY_RSB2C_TO_I 11
EO_ENTWRK_SETINTELM


// find the key in the indexTree
EO_ENT_POSPRIOR
//  EO_ENT_PUSH
//  EO_PUSH_0
//  EO_NOTEQUAL
EO_JMP_FALSE NOT_FOUND

EO_ENT_NEXT
EO_JMP_FALSE NOT_FOUND

EO_ENT_EQUAL
EO_JMP_TRUE FOUND

// NOT_FOUND:
Label NOT_FOUND
EO_ENTWRK_CREATEPAY 88
EO_IXT_INS_ENT
//EO_IXT_DUMP

EO_PAY_ADDRESS

// store Q101_RETURNFLAG -> Q101_RETURNFLAG
EO_RSB_SRC_PUSHCOLS 0
EO_PAY_STORE_COLS 0

// store Q101_LINESTATUS -> Q101_LINESTATUS
EO_RSB_SRC_PUSHCOLS 1
EO_PAY_STORE_COLS 1

// store Q101_SUM_QTY -> Q101_SUM_QTY
EO_RSB_SRC_PUSHCOLI 2
EO_PAY_STORE_COLI 2

// store Q101_SUM_BASE_PRICE -> Q101_SUM_BASE_PRICE
EO_RSB_SRC_PUSHCOLI 3
EO_PAY_STORE_COLI 3

// store Q101_SUM_DISC_PRICE -> Q101_SUM_DISC_PRICE
EO_RSB_SRC_PUSHCOLI 4
EO_PAY_STORE_COLI 4

// store Q101_SUM_CHARGE -> Q101_SUM_CHARGE
EO_RSB_SRC_PUSHCOLI 5
EO_PAY_STORE_COLI 5

// store Q101_COUNT_QTY -> Q101_COUNT_QTY
EO_RSB_SRC_PUSHCOLI 6
EO_PAY_STORE_COLI 6

// store Q101_COUNT_PRICE -> Q101_COUNT_PRICE
EO_RSB_SRC_PUSHCOLI 7
EO_PAY_STORE_COLI 7

// store Q101_SUM_DISC -> Q101_SUM_DISC
EO_RSB_SRC_PUSHCOLI 8
EO_PAY_STORE_COLI 8

// store Q101_COUNT_DISC -> Q101_COUNT_DISC
EO_RSB_SRC_PUSHCOLI 9
EO_PAY_STORE_COLI 9

// store Q101_COUNT_ORDER -> Q101_COUNT_ORDER
EO_RSB_SRC_PUSHCOLI 10
EO_PAY_STORE_COLI 10

//EO_IXT_DUMP

EO_JMP NEXT_RSB_SRC

// FOUND:
Label FOUND

EO_PAY_ADDRESS

// add Q101_SUM_QTY
EO_PAY_FETCH_COLI 2
EO_RSB_SRC_PUSHCOLI 2
EO_ADD
EO_PAY_STORE_COLI 2

// add Q101_SUM_BASE_PRICE
EO_PAY_FETCH_COLI 3
EO_RSB_SRC_PUSHCOLI 3
EO_ADD
EO_PAY_STORE_COLI 3

// add Q101_SUM_DISC_PRICE
EO_PAY_FETCH_COLI 4
EO_RSB_SRC_PUSHCOLI 4
EO_ADD
EO_PAY_STORE_COLI 4

// add Q101_SUM_CHARGE
EO_PAY_FETCH_COLI 5
EO_RSB_SRC_PUSHCOLI 5
EO_ADD
EO_PAY_STORE_COLI 5

// add Q101_COUNT_QTY
EO_PAY_FETCH_COLI 6
EO_RSB_SRC_PUSHCOLI 6
EO_ADD
EO_PAY_STORE_COLI 6

// add Q101_COUNT_PRICE
EO_PAY_FETCH_COLI 7
EO_RSB_SRC_PUSHCOLI 7
EO_ADD
EO_PAY_STORE_COLI 7

// add Q101_SUM_DISC
EO_PAY_FETCH_COLI 8
EO_RSB_SRC_PUSHCOLI 8
EO_ADD
EO_PAY_STORE_COLI 8

// add Q101_COUNT_DISC
EO_PAY_FETCH_COLI 9
EO_RSB_SRC_PUSHCOLI 9
EO_ADD
EO_PAY_STORE_COLI 9

// add Q101_COUNT_ORDER
EO_PAY_FETCH_COLI 10
EO_RSB_SRC_PUSHCOLI 10
EO_ADD
EO_PAY_STORE_COLI 10
//EO_IXT_DUMP

EO_JMP NEXT_RSB_SRC

Label QRY_103_DONE
EO_RSB_SRC_NULL

EO_EXIT
EO_RETURN



// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN SUBROUTINE: SUBROUTINE_EOS (IXT_TO_RSB)
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label SUBROUTINE_EOS

//vAddCodeString(EO_PRINTF **103** SUBROUTINE_IXT_TO_RSB\n

// begin index tree traversal
EO_IXT_TRV_BEGIN
//EO_IXT_DUMP
EO_JMP_FALSE END_IXT_SRC

Label NEXT_IXT_SRC

// new a dst rsb row
EO_RSB_DST_NEW

EO_PAY_ADDRESS

// Q101_RETURNFLAG -> Q104_RETURNFLAG
EO_PAY_FETCH_COLS 0
EO_RSB_DST_POPCOLS 0

// Q101_LINESTATUS -> Q104_LINESTATUS
EO_PAY_FETCH_COLS 1
EO_RSB_DST_POPCOLS 1

//EO_RSB_DST_PRINT

// Q101_SUM_QTY -> Q104_SUM_QTY
EO_PAY_FETCH_COLI 2
EO_RSB_DST_POPCOLI 2

//EO_RSB_DST_PRINT

// Q101_SUM_BASE_PRICE -> Q104_SUM_BASE_PRICE
EO_PAY_FETCH_COLI 3
EO_RSB_DST_POPCOLI 3

// Q101_SUM_DISC_PRICE -> Q104_SUM_DISC_PRICE
EO_PAY_FETCH_COLI 4
EO_RSB_DST_POPCOLI 4

// Q101_SUM_CHARGE -> Q104_SUM_CHARGE
EO_PAY_FETCH_COLI 5
EO_RSB_DST_POPCOLI 5

// Q101_SUM_QTY/Q101_COUNT_QTY -> Q104_AVG_QTY
EO_PAY_FETCH_COLI 2
EO_PAY_FETCH_COLI 6
EO_DIVIDE
EO_RSB_DST_POPCOLI 6

// Q101_SUM_BASE_PRICE/Q101_COUNT_PRICE -> Q104_AVG_PRICE
EO_PAY_FETCH_COLI 3
EO_PAY_FETCH_COLI 7
EO_DIVIDE
EO_RSB_DST_POPCOLI 7

// Q101_SUM_DISC/Q101_COUNT_DISC -> Q104_AVG_DISC
EO_PAY_FETCH_COLI 8
EO_PAY_FETCH_COLI 9
EO_DIVIDE
EO_RSB_DST_POPCOLI 8

// Q101_COUNT_ORDER -> Q104_COUNT_ORDER
EO_PAY_FETCH_COLI 10
EO_RSB_DST_POPCOLI 9

//EO_RSB_DST_PRINT

// NEXT_IXT_SRC:
EO_ENT_NEXT
EO_JMP_TRUE NEXT_IXT_SRC

Label END_IXT_SRC
EO_RETURN

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
