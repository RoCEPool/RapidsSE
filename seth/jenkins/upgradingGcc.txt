curl https://ftp.gnu.org/gnu/gcc/gcc-10.2.0/gcc-10.2.0.tar.gz -O
tar xvzf gcc-10.2.0.tar.gz

# Install prerequisites
sudo yum install gmp-devel mpfr-devel libmpc-devel gcc-c++

mkdir gcc-10.2.0-build
cd gcc-10.2.0-build
../gcc-10.2.0/configure --enable-languages=c,c++ --disable-multilib
make -j$(nproc) 
sudo make install

sudo echo "export PATH=/usr/local/bin:$PATH" >> /etc/profile
sudo echo "export LD_LIBRARY_PATH=/usr/local/lib64:$LD_LIBRARY_PATH" >> /etc/profile
