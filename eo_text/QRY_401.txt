// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN : QRY_401
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Routine QRY_401

  // initialization
  EO_INIT
  EO_ENTER_LU

  // set sql query no (debug)
  EO_SQL 401

  // check for ZK_M_START
  EO_DUP
  EO_INTEGER ZK_M_START
  EO_EQ_INTINT
  EO_JMP_TRUE STARTMARKER_HANDLER

  // check for ZK_M_DATAPULSE
  EO_DUP
  EO_INTEGER ZK_M_DATAPULSE
  EO_EQ_INTINT
  EO_JMP_TRUE DATAPULSE_HANDLER

  // check for ZK_D_STREAMDEF
  EO_DUP
  EO_INTEGER  ZK_D_STREAMDEF
  EO_EQ_INTINT
  EO_JMP_TRUE STREAMDEF_HANDLER

  // check for ZK_D_CRWBUFFER
  EO_DUP
  EO_INTEGER ZK_D_ROWBUFFER
  EO_EQ_INTINT
  EO_JMP_TRUE ROWBUFFER_HANDLER

  // check for ZK_M_EOS
  EO_DUP
  EO_INTEGER  ZK_M_EOS
  EO_EQ_INTINT
  EO_JMP_TRUE EOSMARKER_HANDLER

  // Never reach
  EO_PRINTF "**NO_HANDLER : 401**\n"
  EO_NEVERREACH

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_START handler
Label STARTMARKER_HANDLER
  //EO_PRINTF "**STARTMARKER_HANDLER : 401**\n"
  EO_POP
  //EO_TABLENM_ALT_SET "lineitem"
  EO_PUSH_DEVROW_ALT
  EO_DEVROW_TABLENM_SET "lineitem"
  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_INDEXNM_SET "lineitem_Idx2"

  EO_STRING "idx_vch"
  EO_RLU_INDEXDEF_NEW
  EO_STRING "vchA"
  EO_PUSH_COLUMN_TYPE CT_VARCHAR
  EO_PUSH_RLU_INDEXDEF
  EO_RLU_INDEXDEF_ADDCOLUMN
  EO_PUSH_RLU_INDEXDEF
  EO_RLU_INDEXTREE_NEW
  EO_PUSH_RLU_INDEXDEF
  EO_RLU_INDEXDEF_DELETE

  //EO_INDEXTREE_TMP_SET
  EO_PUSH_RLU_INDEXTREE
  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_INDEXTREE_SET

  EO_JMP QRY_401_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_DATAPULSE handler
Label DATAPULSE_HANDLER
  //EO_PRINTF "**DATAPULSE_HANDLER : 401**\n"
  EO_POP
  EO_JMP QRY_401_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_D_STREAMDEF handler
Label STREAMDEF_HANDLER
  //EO_PRINTF "**STREAMDEF_HANDLER : 401**\n"
  EO_POP

  EO_REG_STREAMDEFOUT_NEW

  // add column o_orderpriorty
  EO_PUSH_COLUMN_SCALE -1
  EO_PUSH_COLUMN_LENGTH 15
  EO_PUSH_COLUMN_TYPE CT_VARCHAR
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderpriorty"

  // add column order_count
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "order_count"

  EO_POT_USE_STREAMDEFOUT 1

  EO_JMP QRY_401_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_EOS handler
Label EOSMARKER_HANDLER
  //EO_PRINTF "**EOSMARKER_HANDLER : 401**\n"
  EO_POP
  EO_CALL OUT_AGG

  EO_JMP QRY_401_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_D_ROWBUFFER handler
Label ROWBUFFER_HANDLER
  //EO_PRINTF "**ROWBUFFER_HANDLER : 401**\n"
  EO_POP

  // forearch src rsb
  //EO_ROW_SRC_BEGIN
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ROWBUF_BEGIN

Label NEXT_RSB_SRC

  //EO_ROW_SRC_NEXT
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ROWBUF_NEXT

  EO_JMP_FALSE QRY_401_DONE

  // test for a matching LINEITEM
  EO_CALL TST_LINEITEM
  EO_JMP_FALSE NEXT_RSB_SRC

  // get the aggregator for o_orderpriority
  EO_CALL GET_AGR

  EO_PUSH_DEVIXT_TMP
  EO_PAY_ADDRESS
  EO_PAY_FETCH_COLI 0
  EO_INC
  EO_PAY_STORE_COLI 0
  EO_JMP NEXT_RSB_SRC


Label QRY_401_DONE
  EO_EXIT_LU
  EO_RETURN

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// END MAIN: QRY_401
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: TST_LINEITEM -- test for matching lineitem
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label TST_LINEITEM

  //EO_ROW_SRC_PRINT
  EO_PUSH_RLU_ENTWRK
  EO_ENTWRK_RESET

  //EO_ROW_SRC_PUSHCOLI 0 // O_ORDERKEY
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_PUSHCOLI 0 // O_ORDERKEY

  EO_PUSH_RLU_ENTWRK
  EO_ENTWRK_SETINTELM_OLD

  // position / iterate
  EO_PUSH_DEVIXT_ALT
  EO_PUSH_RLU_ENT
  EO_DEVIXT_POS_BYKEY

  EO_JMP_FALSE RET_LINEITEM_f

Label NXT_ITM
  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_IDX_NEXT
  EO_JMP_FALSE RET_LINEITEM_f
  //EO_IXT_ALT_PRINT

  EO_PUSH_DEVIXT_ALT
  EO_PUSH_DEVIXT_ENT
  EO_PUSH_RLU_ENT
  EO_DEVIXT_CMP_KEYS
  EO_JMP_FALSE RET_LINEITEM_f

  // test L_COMMITDATE < L_RECEIPTDATE
  //EO_CRW_ALT_DEREF
  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_ENT_DEREF

  //EO_COL_ALT_PUSHONE L_LINENUMBER
  //EO_COL_ALT_PUSHONE L_ORDERKEY
  //EO_PRINTF "  joined L_OrderKey %i, L_LineNumber %i\n"_Stz

  EO_RLU_SET_ITEMS 2
  EO_RLU_SET_COLUMN 12 // L_RECIPTDATE
  EO_RLU_SET_COLUMN 11 // L_COMMITDATE
  EO_PUSH_RLU_COLSET
  EO_PUSH_DEVROW_ALT
  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_COL_PUSHSET_PST

  EO_LT_INTINT
  EO_JMP_FALSE NXT_ITM
  EO_JMP RET_LINEITEM_t

  EO_JMP NXT_ITM

  // found one matching lineitem
Label RET_LINEITEM_t
  EO_PUSH_1
  EO_RETURN

  // no matching lineitem
Label RET_LINEITEM_f
  EO_PUSH_0
  EO_RETURN

// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: RET_ITM
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: GET_AGR
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// get aggregator for o_orderpriority
Label GET_AGR

  // reset entwrk
  EO_PUSH_RLU_ENTWRK
  EO_ENTWRK_RESET

  //EO_ROW_SRC_PRINT
  //EO_ROW_SRC_PUSHCOLS 1 //O_ORDERPRIORITY
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_PUSHCOLS 1 //O_ORDERPRIORITY

  EO_PUSH_RLU_ENTWRK
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ENTWRK_APPENDELMS

  // attempt to find the key in the indexTree
  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_ENT_POSPRIOR
  EO_JMP_FALSE NEW_AGR

  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_ENT_NEXT
  EO_JMP_FALSE NEW_AGR

  //EO_PUSH_DEVIXT_TMP
  //EO_PUSH_RLU_ENT
  //EO_DEVIXT_ENT_PRINTKEYS

  EO_PUSH_DEVIXT_TMP
  EO_PUSH_RLU_ENT
  EO_DEVIXT_ENTKEY_EQUAL

  EO_JMP_FALSE NEW_AGR

  //EO_IXT_DUMP,(i8)0

  // found the agg
  EO_JMP END_AGR

  // did not find agg, new one
  Label NEW_AGR
  //EO_ENTWRK_CREATEPAY 8 //1 * C_BYTES_PER_WORD
  EO_PUSH_RLU_ENTWRK
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ENTWRK_CREATEPAY 8 //1 * C_BYTES_PER_WORD

  EO_PUSH_DEVIXT_TMP
  EO_PUSH_RLU_ENT
  EO_DEVIXT_IDX_INS

  EO_PUSH_DEVIXT_TMP
  EO_PAY_ADDRESS

  EO_PUSH_0
  EO_PAY_STORE_COLI 0

  //EO_IXT_DUMP,(i8)0

Label END_AGR
  EO_RETURN
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: GET_AGR                                        Q400 GET_AGR
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: OUT_AGG                                            Q400 OUT_AGG
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label OUT_AGG

  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_IDX_BEGIN

  EO_JMP_FALSE TRV_RET

  Label TRV_ECH

  //EO_ROW_DST_NEW
  EO_PUSH_DEVROW_DST
  EO_PUSH_RLU_POT
  EO_PUSH_RLU_PTN
  EO_DEVROW_ROW_NEW

  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_ENT_PUSHELMS0
  //EO_ROW_DST_POPCOLS 0 // Q400_ORDERPRIORITY
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLS 0

  EO_PUSH_DEVIXT_TMP
  EO_PAY_ADDRESS

  EO_PAY_FETCH_COLI 0
  //EO_ROW_DST_POPCOLI 1 // Q400_ORDERCOUNT
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLI 1 // Q400_ORDERCOUNT

  //EO_ROW_DST_PRINT

  Label TRV_NXT
  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_IDX_NEXT
  EO_JMP_TRUE TRV_ECH

  Label TRV_RET
  EO_RETURN
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: OUT_AGG                                        Q400 OUT_AGG
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
