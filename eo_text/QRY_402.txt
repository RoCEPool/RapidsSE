// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN : QRY_402
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Routine QRY_402

  // initialization
  EO_INIT
  EO_ENTER_LU

  // set sql query no (debug)
  EO_SQL 402

  // check for ZK_M_START
  EO_DUP
  EO_INTEGER ZK_M_START
  EO_EQ_INTINT
  EO_JMP_TRUE STARTMARKER_HANDLER

  // check for ZK_M_DATAPULSE
  EO_DUP
  EO_INTEGER ZK_M_DATAPULSE
  EO_EQ_INTINT
  EO_JMP_TRUE DATAPULSE_HANDLER

  // check for ZK_D_STREAMDEF
  EO_DUP
  EO_INTEGER  ZK_D_STREAMDEF
  EO_EQ_INTINT
  EO_JMP_TRUE STREAMDEF_HANDLER

  // check for ZK_D_ROWBUFFER
  EO_DUP
  EO_INTEGER ZK_D_ROWBUFFER
  EO_EQ_INTINT
  EO_JMP_TRUE ROWBUFFER_HANDLER

  // check for ZK_M_EOS
  EO_DUP
  EO_INTEGER  ZK_M_EOS
  EO_EQ_INTINT
  EO_JMP_TRUE EOSMARKER_HANDLER

  // Never reach
  EO_PRINTF "**NO_HANDLER : 402**\n"
  EO_NEVERREACH

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_START handler
Label STARTMARKER_HANDLER
  //EO_PRINTF "**STARTMARKER_HANDLER : 402**\n"
  EO_POP
  EO_STRING "idx_vch"
  EO_RLU_INDEXDEF_NEW
  EO_STRING "vchA"
  EO_PUSH_COLUMN_TYPE CT_VARCHAR
  EO_PUSH_RLU_INDEXDEF
  EO_RLU_INDEXDEF_ADDCOLUMN
  EO_PUSH_RLU_INDEXDEF
  EO_RLU_INDEXTREE_NEW
  EO_PUSH_RLU_INDEXDEF
  EO_RLU_INDEXDEF_DELETE

  //EO_INDEXTREE_TMP_SET
  EO_PUSH_RLU_INDEXTREE
  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_INDEXTREE_SET

  EO_JMP QRY_402_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_DATAPULSE handler
Label DATAPULSE_HANDLER
  //EO_PRINTF "**DATAPULSE_HANDLER : 402**\n"
  EO_POP
  EO_JMP QRY_402_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_D_STREAMDEF handler
Label STREAMDEF_HANDLER
  //EO_PRINTF "**STREAMDEF_HANDLER : 402**\n"
  EO_POP

  EO_REG_STREAMDEFOUT_NEW

  // add column o_orderpriorty
  EO_PUSH_COLUMN_SCALE -1
  EO_PUSH_COLUMN_LENGTH 15
  EO_PUSH_COLUMN_TYPE CT_VARCHAR
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderpriorty"

  // add column order_count
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "order_count"

  EO_POT_USE_STREAMDEFOUT 1

  EO_JMP QRY_402_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_EOS handler
Label EOSMARKER_HANDLER
  //EO_PRINTF "**EOSMARKER_HANDLER : 402**\n"
  EO_POP

  EO_CALL OUT_AGG

  EO_JMP QRY_402_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_D_ROWBUFFER handler
Label ROWBUFFER_HANDLER
  //EO_PRINTF "**ROWBUFFER_HANDLER : 402**\n"
  EO_POP

  EO_CALL ITR_SRC
  EO_JMP QRY_402_DONE

Label QRY_402_DONE
  EO_EXIT_LU
  EO_RETURN


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
//                                                                Q401 ITR_SRC
Label ITR_SRC

  //EO_ROW_SRC_BEGIN
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ROWBUF_BEGIN

Label NXT_SRC
  //EO_ROW_SRC_NEXT
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ROWBUF_NEXT

  EO_JMP_FALSE ITR_END

  //EO_ROW_SRC_PRINT

  EO_CALL GET_AGR

  EO_CALL SUP_AGR

  EO_JMP NXT_SRC

// --
Label ITR_END
  EO_RETURN


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
//                                                                Q401 SUP_AGR
Label SUP_AGR

  EO_PUSH_DEVIXT_TMP
  EO_PAY_ADDRESS

  //EO_ROW_SRC_PUSHCOLI 1 //Q400_ORDERCOUNT
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_PUSHCOLI 1 //Q400_ORDERCOUNT

  EO_PAY_FETCH_COLI 0
  EO_ADD_INTINT
  EO_PAY_STORE_COLI 0

Label SUP_END
  EO_RETURN

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
//                                                                Q401 GET_AGR
// get aggregator for o_orderpriority
Label GET_AGR

// reset entwrk
  EO_PUSH_RLU_ENTWRK
  EO_ENTWRK_RESET

// add key O_ORDERPRIORITY
  //EO_ROW_SRC_PUSHCOLS 0 //Q400_ORDERPRIORITY
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_PUSHCOLS 0 //Q400_ORDERPRIORITY

  EO_PUSH_RLU_ENTWRK
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ENTWRK_APPENDELMS

  // attempt to find the key in the indexTree
  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_ENT_POSPRIOR
  EO_JMP_FALSE NEW_AGR

  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_ENT_NEXT
  EO_JMP_FALSE NEW_AGR

  EO_PUSH_DEVIXT_TMP
  EO_PUSH_RLU_ENT
  EO_DEVIXT_ENTKEY_EQUAL

  EO_JMP_FALSE NEW_AGR

//EO_IXT_DUMP,(i8)0

// found the agg
  EO_JMP END_AGR

// did not find agg, new one
Label NEW_AGR

  //EO_ENTWRK_CREATEPAY 8 //1 * C_BYTES_PER_WORD
  EO_PUSH_RLU_ENTWRK
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ENTWRK_CREATEPAY 8 //1 * C_BYTES_PER_WORD

  EO_PUSH_DEVIXT_TMP
  EO_PUSH_RLU_ENT
  EO_DEVIXT_IDX_INS

  EO_PUSH_DEVIXT_TMP
  EO_PAY_ADDRESS

  EO_PUSH_0
  EO_PAY_STORE_COLI 0

//EO_IXT_DUMP,(i8)0

Label END_AGR
  EO_RETURN

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
//
Label OUT_AGG
  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_IDX_BEGIN

  EO_JMP_FALSE TRV_RET

Label TRV_ECH

  //EO_ROW_DST_NEW
  EO_PUSH_DEVROW_DST
  EO_PUSH_RLU_POT
  EO_PUSH_RLU_PTN
  EO_DEVROW_ROW_NEW

  EO_PUSH_DEVIXT_TMP
  EO_PAY_ADDRESS

  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_ENT_PUSHELMS0
  //EO_ROW_DST_POPCOLS 0 //Q401_ORDERPRIORITY
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLS 0

  EO_PAY_FETCH_COLI 0
  //EO_ROW_DST_POPCOLI 1 //Q401_ORDERCOUNT
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLI 1

  //EO_ROW_DST_PRINT

Label TRV_NXT
  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_IDX_NEXT
  EO_JMP_TRUE TRV_ECH

Label TRV_RET
  EO_RETURN
