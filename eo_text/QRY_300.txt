// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN : QRY_300
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Routine QRY_300

  // initialization
  EO_INIT
  EO_ENTER_LU

  // set sql query no (debug)
  EO_SQL 300

  // check for ZK_M_START
  EO_DUP
  EO_INTEGER ZK_M_START
  EO_EQ_INTINT
  EO_JMP_TRUE STARTMARKER_HANDLER

  // check for ZK_M_DATAPULSE
  EO_DUP
  EO_INTEGER ZK_M_DATAPULSE
  EO_EQ_INTINT
  EO_JMP_TRUE DATAPULSE_HANDLER

  // check for ZK_D_STREAMDEF
  EO_DUP
  EO_INTEGER  ZK_D_STREAMDEF
  EO_EQ_INTINT
  EO_JMP_TRUE STREAMDEF_HANDLER

  // check for ZK_M_EOS
  EO_DUP
  EO_INTEGER ZK_M_EOS
  EO_EQ_INTINT
  EO_JMP_TRUE EOSMARKER_HANDLER

  // Never reach
  EO_PRINTF "**NO_HANDLER : 300**\n"
  EO_NEVERREACH

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_M_START handler
Label STARTMARKER_HANDLER
  EO_PRINTF "**STARTMARKER_HANDLER : 300**\n"
  EO_POP

  //EO_TABLENM_SRC_SET "customer"
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_TABLENM_SET "customer"

  //EO_INDEXNM_SRC_SET "customer_Idx1"
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_INDEXNM_SET "customer_Idx1"

  //EO_TABLENM_ALT_SET "orders"
  EO_PUSH_DEVROW_ALT
  EO_DEVROW_TABLENM_SET "orders"

  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_INDEXNM_SET "orders_Idx1"

  EO_REG_STREAMDEFOUT_NEW

  // add column o_orderkey
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderkey"

  // add column o_orderdate
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 0
  EO_PUSH_COLUMN_TYPE CT_DATE
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderdate"

  // add column o_shippriority
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_shippriority"

  EO_POT_USE_STREAMDEFOUT 0
  EO_JMP QRY_300_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_D_STREAMDEF handler
Label STREAMDEF_HANDLER
  EO_PRINTF "**STREAMDEF_HANDLER : 300**\n"
  EO_POP
  EO_JMP QRY_300_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_M_EOS handler
Label EOSMARKER_HANDLER
  EO_PRINTF "**EOSMARKER_HANDLER : 300**\n"
  EO_POP
  EO_JMP QRY_300_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_M_DATAPULSE handler
Label DATAPULSE_HANDLER
  EO_PRINTF "**DATAPULSE_HANDLER : 300**\n"
  EO_POP

  EO_PUSH_0
  EO_CTR_0_POP

  // test if rtn has been called again
  EO_CALLEDAGAIN
  EO_JMP_TRUE PRC_CUS

  //EO_INTEGRITY;


  // iterate all CUSTOMER
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_IDX_BEGIN
  EO_JMP_TRUE GOT_CUS

  EO_JMP QRY_300_DONE

Label GOT_CUS
  //EO_PRINTF "**QRY_300 GOT_CUS**\n"
  EO_CTR_0_INC
  EO_CTR_0_PUSH
  EO_INTEGER 10000
  EO_LT_INTINT
  EO_JMP_TRUE PRC_CUS

  EO_PRINTF "**QRY_300 TO BE CALLED AGAIN**\n"
  EO_CALLAGAIN
  EO_EXIT_LU
  EO_RETURN

Label PRC_CUS
  //EO_CRW_SRC_DEREF
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_ENT_DEREF

  //EO_COL_SRC_PUSHONE_a 6 //C_MKTSEGMENT
  EO_RLU_SET_ITEMS 1
  EO_RLU_SET_COLUMN 6 //C_MKTSEGMENT
  EO_PUSH_RLU_COLSET
  EO_PUSH_DEVROW_LFT
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_COL_PUSHSET_ADR

  EO_STR_LITERAL <param0>
  EO_STR_COMPARE_a
  EO_JMP_NOTEQUAL NXT_CUSTOMER

////
//EO_COL_SRC_PUSHONE 0
//EO_PRINTF "Next C_CustKey %i\n"
////

  EO_CALL JOI_ORD


// next CUSTOMER
Label NXT_CUSTOMER
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_IDX_NEXT
  EO_JMP_TRUE GOT_CUS

  EO_REG_EOSMARKER_NEW

// end of CUSTOMER
Label QRY_300_DONE
  EO_EXIT_LU
  EO_RETURN

// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END ROUTINE: QRY_300                                           Q300
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: JOI_ORD  -- join with orders                       Q300 JOI_ORD
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label JOI_ORD
  //EO_PRINTF "*** JOI_ORD ***\n"
  EO_PUSH_DEVIXT_ALT
  EO_PUSH_DEVIXT_LFT
  EO_PUSH_DEVIXT_ENT
  EO_DEVIXT_POS_BYKEY
  EO_JMP_FALSE RET_ORD

Label NXT_ORD
  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_IDX_NEXT

  EO_JMP_FALSE RET_ORD

  EO_PUSH_DEVIXT_LFT
  EO_PUSH_DEVIXT_ENT
  EO_PUSH_DEVIXT_ALT
  EO_PUSH_DEVIXT_ENT
  EO_DEVIXT_CMP_KEYS

  EO_JMP_FALSE RET_ORD

  //EO_CRW_ALT_DEREF
  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_ENT_DEREF

  EO_RLU_SET_ITEMS 1
  EO_RLU_SET_COLUMN 4 //O_ORDERDATE
  EO_PUSH_RLU_COLSET
  EO_PUSH_DEVROW_ALT
  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_COL_PUSHSET_PST
  EO_DATE  <param1>
  EO_LT_INTINT
  EO_JMP_FALSE NXT_ORD

/*
  EO_COL_ALT_PUSHONE 4 //O_ORDERDATE
  EO_DATE  "1994-11-14"
  EO_GE_INTINT
  EO_JMP_FALSE NXT_ORD

  EO_COL_ALT_PUSHONE 4 //O_ORDERDATE
  EO_DATE  <param1>
  EO_LT_INTINT
  EO_JMP_FALSE NXT_ORD

*/
////
//EO_COL_ALT_PUSHONE 0
//EO_PRINTF "joined O_OrderKey %i\n"
////

  EO_CALL DST_PUT

  EO_JMP NXT_ORD

Label RET_ORD
  EO_RETURN
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: JOI_ORD                                        Q300 JOI_ORD
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =


// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: DST_PUT                                            Q300 DST_PUT
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label DST_PUT

  // calc i_Ptn
  EO_PTN_RESET

  EO_RLU_SET_ITEMS 1
  EO_RLU_SET_COLUMN 0 //O_ORDERKEY
  EO_PUSH_RLU_COLSET
  EO_PUSH_DEVROW_ALT
  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_COL_PUSHSET_PST
  EO_PTN_RUNNING
  EO_PTN_CALC

  EO_RLU_SET_ITEMS 3
  EO_RLU_SET_COLUMN 7 //O_SHIPPRIORITY  // 7
  EO_RLU_SET_COLUMN 4 //O_ORDERDATE     // 4
  EO_RLU_SET_COLUMN 0 //O_ORDERKEY      // 0
  EO_PUSH_RLU_COLSET
  EO_PUSH_DEVROW_ALT
  EO_PUSH_DEVIXT_ALT
  EO_DEVIXT_COL_PUSHSET_PST

  //EO_ROW_DST_NEW
  EO_PUSH_DEVROW_DST
  EO_PUSH_RLU_POT
  EO_PUSH_RLU_PTN
  EO_DEVROW_ROW_NEW

  //EO_ROW_DST_POPCOLI 2 //Q300_SHIPPRIORITY
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLI 2

  //EO_ROW_DST_POPCOLI 1 //Q300_ORDERDATE
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLI 1

  //EO_ROW_DST_POPCOLI 0 //Q300_ORDERKEY
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLI 0

  //EO_ROW_DST_PRINT
  EO_RETURN

// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: DST_PUT                                        Q300 DST_PUT
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
