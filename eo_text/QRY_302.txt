Routine QRY_302

  // set sql query no (debug)
  EO_SQL 302

  EO_DUP
  EO_INTEGER ZK_M_START
  EO_EQ_INTINT
  EO_JMP_TRUE STARTMARKER_HANDLER

  EO_JMP_TRUE ROWBUFFER_HANDLER

  // check for ZK_M_EOS

  EO_INTEGER  ZK_M_EOS
  EO_EQ_INTINT
  EO_JMP_TRUE EOSMARKER_HANDLER

  // Never reach
  EO_PRINTF "**NO_HANDLER : 302**\n"
  EO_NEVERREACH

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_START handler
Label STARTMARKER_HANDLER
  EO_PRINTF "**STARTMARKER_HANDLER : 302**\n"
  EO_POP
  EO_STRING "idx_int"
  EO_RLU_INDEXDEF_NEW
  EO_STRING "intA"
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_PUSH_RLU_INDEXDEF
  EO_RLU_INDEXDEF_ADDCOLUMN
  EO_PUSH_RLU_INDEXDEF
  EO_RLU_INDEXTREE_NEW
  EO_PUSH_RLU_INDEXDEF
  EO_RLU_INDEXDEF_DELETE

  //EO_INDEXTREE_TMP_SET
  EO_PUSH_RLU_INDEXTREE
  EO_PUSH_DEVIXT_TMP

  EO_PRINTF "**DATAPULSE_HANDLER : 302**\n"

  EO_JMP QRY_302_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

Label STREAMDEF_HANDLER
  EO_PRINTF "**STREAMDEF_HANDLER : 302**\n"
  EO_POP

  EO_REG_STREAMDEFOUT_NEW

  // add column l_orderkey
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "l_orderkey"

  // add column revenue
  EO_PUSH_COLUMN_SCALE 2
  EO_PUSH_COLUMN_LENGTH 17
  EO_PUSH_COLUMN_TYPE CT_DECIMAL
  EO_REG_STREAMDEFOUT_ADDCOLUMN "revenue"

  // add column o_orderdate
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 0
  EO_PUSH_COLUMN_TYPE CT_DATE
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderdate"

  // add column o_shippriority
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_shippriority"


  EO_POT_USE_STREAMDEFOUT 0

  EO_JMP QRY_302_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_EOS handler
Label EOSMARKER_HANDLER
  EO_PRINTF "**EOSMARKER_HANDLER : 302**\n"
  EO_POP
  EO_CALL OUT_AGG
  EO_JMP QRY_END
  EO_JMP QRY_302_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_D_ROWBUFFER handler
Label ROWBUFFER_HANDLER
  EO_PRINTF "**ROWBUFFER_HANDLER : 302**\n"

  // --
  //EO_ROW_SRC_BEGIN
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ROWBUF_BEGIN

Label NXT_SRC
  //EO_ROW_SRC_NEXT
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ROWBUF_NEXT

  EO_JMP_FALSE QRY_302_DONE

  //EO_ROW_SRC_PRINT
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ROW_PRINT

  //EO_CALL,"CPY_DST
  EO_CALL LIM_SBR

  EO_JMP NXT_SRC


// --
Label QRY_302_DONE
EO_EXIT_LU
EO_RETURN

// --
Label QRY_END

EO_EXIT_LU
EO_RETURN

// END ROUTINE: QRY_302                                           Q302

  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_ENT_PUSHELMI0

  EO_PUSH_RLU_ENTWRK

  //EO_ROW_SRC_PUSHCOLI 1 //Q301_REVENUE
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_PUSHCOLI 1 //Q301_REVENUE

  //EO_DUP
  //EO_PRINT
  //EO_ENTWRK_APPENDELMI
  EO_PUSH_RLU_ENTWRK
  EO_ENTWRK_SETINTELM_OLD

  //EO_ENTWRK_CREATEPAY 24 //3 * C_BYTES_PER_WORD

  EO_PAY_ADDRESS

  //EO_ROW_SRC_PUSHCOLI 0 //Q301_ORDERKEY

  EO_PUSH_DEVROW_LFT

  EO_PAY_STORE_COLI 1

  EO_DEVROW_PUSHCOLI 3 //Q301_SHIPPRIORITY

  EO_DUP

  //EO_IXT_TMP_ENTRIES
  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_IDX_ENTRIES

  EO_JMP_FALSE TRV_BEG

  EO_ADD_INTINT

  //EO_ROW_DST_NEW

  EO_PUSH_RLU_POT

  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_IDX_BEGIN

  EO_JMP_FALSE TRV_RET

  //EO_IXT_TMP_ENTRIES
  EO_PUSH_DEVIXT_TMP
  EO_DEVIXT_IDX_ENTRIES

  EO_SUB_INTINT

  EO_DUP

  EO_PUSH_DEVROW_DST
  EO_DEVROW_ROW_SET

  EO_PUSH_DEVIXT_TMP
  EO_PAY_ADDRESS
