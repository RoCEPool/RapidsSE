// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN: QRY_601
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Routine QRY_601

  EO_INIT
  EO_ENTER_LU

  EO_SQL 601

  // check for ZK_M_START
  EO_DUP
  EO_INTEGER  ZK_M_START
  EO_EQ_INTINT
  EO_JMP_TRUE STARTMARKER_HANDLER

  // check for PULSEMARKER
  EO_DUP
  EO_INTEGER  ZK_M_DATAPULSE
  EO_EQ_INTINT
  EO_JMP_TRUE DATAPULSE_HANDLER

  // check for ZK_D_STREAMDEF
  EO_DUP
  EO_INTEGER  ZK_D_STREAMDEF
  EO_EQ_INTINT
  EO_JMP_TRUE STREAMDEF_HANDLER

  // check for ZK_M_EOS
  EO_DUP
  EO_INTEGER ZK_M_EOS
  EO_EQ_INTINT
  EO_JMP_TRUE EOSMARKER_HANDLER

  // check for ZK_D_ROWBUFFER
  EO_DUP
  EO_INTEGER ZK_D_ROWBUFFER
  EO_EQ_INTINT
  EO_JMP_TRUE ROWBUFFER_HANDLER

  EO_PRINTF "**NO_HANDLER : 601**\n"
  EO_NEVERREACH

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_START handler
Label STARTMARKER_HANDLER
  EO_PRINTF "**STARTMARKER_HANDLER : 601**\n"
  EO_POP
  EO_JMP QRY_601_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_DATAPULSE handler
Label DATAPULSE_HANDLER
  EO_PRINTF "**DATAPULSE_HANDLER : 601**\n"
  EO_POP
  EO_JMP QRY_601_DONE
  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_D_STREAMDEF handler
Label STREAMDEF_HANDLER
  EO_PRINTF "**STREAMDEF_HANDLER : 601**\n"
  EO_POP
  EO_REG_STREAMDEF_STREAMDEFOUT
  EO_POT_USE_STREAMDEFOUT 1
  EO_JMP QRY_601_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_EOS handler
Label EOSMARKER_HANDLER
  EO_PRINTF "**EOSMARKER_HANDLER : 601**\n"
  EO_POP
  EO_JMP QRY_601_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_D_ROWBUFFER handler
Label ROWBUFFER_HANDLER
  EO_PRINTF "**ROWBUFFER_HANDLER : 601**\n"
  EO_POP

  //EO_ROW_DST_TESTROW
  EO_PUSH_DEVROW_DST
  EO_DEVROW_ROW_TEST

  EO_JMP_TRUE RSB_SRC_BEGIN

  //EO_ROW_DST_NEW
  EO_PUSH_DEVROW_DST
  EO_PUSH_RLU_POT
  EO_PUSH_RLU_PTN
  EO_DEVROW_ROW_NEW

  EO_PUSH_0
  //EO_ROW_DST_POPCOLI 0
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLI 0

Label RSB_SRC_BEGIN
  //EO_ROW_SRC_BEGIN
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ROWBUF_BEGIN

Label NEXT_RSB_SRC
  //EO_ROW_SRC_NEXT
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_ROWBUF_NEXT

  EO_JMP_FALSE QRY_601_DONE
  EO_CALL SUBROUTINE_AGGREGATE
  EO_JMP NEXT_RSB_SRC

Label QRY_601_DONE
  EO_RETURN

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// END: QRY_601
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN: SUBROUTINE_AGGREGATE
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label SUBROUTINE_AGGREGATE
  //EO_ROW_SRC_PUSHCOLI 0
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_PUSHCOLI 0

  //EO_ROW_DST_PUSHCOLI 0
  EO_PUSH_DEVROW_DST
  EO_DEVROW_PUSHCOLI 0

  EO_ADD_INTINT
  //EO_ROW_DST_POPCOLI 0
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLI 0

  EO_RETURN
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// END: SUBROUTINE_AGGREGATE
