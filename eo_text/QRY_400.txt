
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// BEGIN : QRY_400
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Routine QRY_400

  // initialization
  EO_INIT
  EO_ENTER_LU

  // set sql query no (debug)
  EO_SQL 400

  // check for ZK_M_START
  EO_DUP
  EO_INTEGER ZK_M_START
  EO_EQ_INTINT
  EO_JMP_TRUE STARTMARKER_HANDLER

  // check for ZK_M_DATAPULSE
  EO_DUP
  EO_INTEGER ZK_M_DATAPULSE
  EO_EQ_INTINT
  EO_JMP_TRUE DATAPULSE_HANDLER

  // check for ZK_D_STREAMDEF
  EO_DUP
  EO_INTEGER  ZK_D_STREAMDEF
  EO_EQ_INTINT
  EO_JMP_TRUE STREAMDEF_HANDLER

  // check for ZK_M_EOS
  EO_DUP
  EO_INTEGER  ZK_M_EOS
  EO_EQ_INTINT
  EO_JMP_TRUE EOSMARKER_HANDLER

  // Never reach
  EO_PRINTF "**NO_HANDLER : 400**\n"
  EO_NEVERREACH

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_START handler
Label STARTMARKER_HANDLER
  //EO_PRINTF "**STARTMARKER_HANDLER : 400**\n"
  EO_POP

  //EO_TABLENM_SRC_SET "orders"
  EO_PUSH_DEVROW_LFT
  EO_DEVROW_TABLENM_SET "orders"

  //EO_INDEXNM_SRC_SET "orders_Idx2"
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_INDEXNM_SET "orders_Idx2"

  EO_JMP QRY_400_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_M_EOS handler
Label EOSMARKER_HANDLER
  //EO_PRINTF "**EOSMARKER_HANDLER : 400**\n"
  EO_POP
  EO_JMP QRY_400_DONE

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// ZK_D_STREAMDEF handler
Label STREAMDEF_HANDLER
  //EO_PRINTF "**STREAMDEF_HANDLER : 400**\n"
  EO_POP

  EO_JMP QRY_400_DONE

  // - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
  // ZK_M_DATAPULSE handler
Label DATAPULSE_HANDLER
  //EO_PRINTF "**DATAPULSE_HANDLER : 400**\n"
  EO_POP

  EO_REG_STREAMDEFOUT_NEW

  // add column o_orderkey
  EO_PUSH_COLUMN_SCALE 0
  EO_PUSH_COLUMN_LENGTH 19
  EO_PUSH_COLUMN_TYPE CT_INTEGER
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderkey"

  // add column o_orderpriorty
  EO_PUSH_COLUMN_SCALE -1
  EO_PUSH_COLUMN_LENGTH 15
  EO_PUSH_COLUMN_TYPE CT_VARCHAR
  EO_REG_STREAMDEFOUT_ADDCOLUMN "o_orderpriorty"

  EO_POT_USE_STREAMDEFOUT 1

  // iterate all ORDERS
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_IDX_BEGIN
  EO_JMP_FALSE QRY_400_DONE

Label GOT_ORD
  //EO_CRW_SRC_DEREF
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_ENT_DEREF

  //vCC(1); vAddCodeArg(EO_COL_SRC_PUSHONE,O_ORDERKEY);
  //vCC(1); vAddCodeString(EO_PRINTF,"Next O_OrderKey %i\n"_Stz);

  // test O_ORDERDATE
  EO_RLU_SET_ITEMS 1
  EO_RLU_SET_COLUMN 4

  EO_PUSH_RLU_COLSET
  EO_PUSH_DEVROW_LFT
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_COL_PUSHSET_PST
  EO_DUP
  EO_DATE <param0> //"1993-07-01"
  EO_GE_INTINT
  EO_JMP_TRUE TST_2ND
  EO_POP
  EO_JMP NXT_ORD

Label TST_2ND
  EO_DATE <param1> //"1993-10-01"
  EO_LT_INTINT
  EO_JMP_FALSE NXT_ORD

  EO_CALL DST_PUT

Label NXT_ORD
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_IDX_NEXT
  EO_JMP_TRUE GOT_ORD
  EO_REG_EOSMARKER_NEW

Label QRY_400_DONE
  EO_RETURN

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// END MAIN: QRY_400
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
// SUBROUTINE: DST_PUT                                            Q300 DST_PUT
// - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
Label DST_PUT
  //EO_PRINTF "DST_PUT\n"

  // calc i_Ptn
  EO_PTN_RESET

  EO_RLU_SET_ITEMS 1
  EO_RLU_SET_COLUMN 0 //O_ORDERKEY
  EO_PUSH_RLU_COLSET
  EO_PUSH_DEVROW_LFT
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_COL_PUSHSET_PST
  EO_PTN_RUNNING
  EO_PTN_CALC

  EO_RLU_SET_ITEMS 2
  EO_RLU_SET_COLUMN 5 //O_ORDERPRIORITY
  EO_RLU_SET_COLUMN 0 //O_ORDERKEY
  EO_PUSH_RLU_COLSET
  EO_PUSH_DEVROW_LFT
  EO_PUSH_DEVIXT_LFT
  EO_DEVIXT_COL_PUSHSET_PST

  //EO_ROW_DST_NEW
  EO_PUSH_DEVROW_DST
  EO_PUSH_RLU_POT
  EO_PUSH_RLU_PTN
  EO_DEVROW_ROW_NEW

  //EO_ROW_DST_POPCOLS 1 //O_ORDERPRIORITY
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLS 1

  //EO_ROW_DST_POPCOLI 0 //Q300_ORDERKEY
  EO_PUSH_DEVROW_DST
  EO_DEVROW_POPCOLI 0

  //EO_ROW_DST_PRINT
  EO_RETURN

// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
// END SUBROUTINE: DST_PUT                                        Q300 DST_PUT
// = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = =
